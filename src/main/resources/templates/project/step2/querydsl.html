CommonQueryConditions 모듈

// 공통 조건 모듈 public class CommonQueryConditions { // 삭제되지 않은 데이터 public static BooleanExpression notDeleted(BooleanPath deletedPath) { return deletedPath.eq(false); } // 문자열 포함 검색 public static BooleanExpression stringContains(StringExpression stringPath, String keyword) { if (keyword == null || keyword.trim().isEmpty()) return null; return stringPath.contains(keyword.trim()); } // Enum 값 비교 public static <T extends Enum<T>> BooleanExpression enumEq(EnumPath<T> enumPath, T value) { return value != null ? enumPath.eq(value) : null; } // 날짜 범위 검색 public static BooleanExpression dateBetween(DateTimePath<LocalDateTime> dateTimePath, LocalDateTime start, LocalDateTime end) { if (start != null && end != null) { return dateTimePath.between(start, end); } else if (start != null) { return dateTimePath.goe(start); } else if (end != null) { return dateTimePath.loe(end); } return null; } // 숫자 범위 검색 public static <T extends Number & Comparable<?>> BooleanExpression numberBetween(NumberPath<T> numberPath, T min, T max) { if (min != null && max != null) { return numberPath.between(min, max); } else if (min != null) { return numberPath.goe(min); } else if (max != null) { return numberPath.loe(max); } return null; } // 비율 계산 (수강률 등) public static BooleanExpression ratioBetween(NumberPath<Integer> numerator, NumberPath<Integer> denominator, Double minRatio, Double maxRatio) { if (numerator == null || denominator == null || denominator.eq(0).equals(true)) { return null; } BooleanExpression condition = denominator.gt(0); if (minRatio != null && maxRatio != null) { condition = condition.and(numerator.divide(denominator).between(minRatio, maxRatio)); } else if (minRatio != null) { condition = condition.and(numerator.divide(denominator).goe(minRatio)); } else if (maxRatio != null) { condition = condition.and(numerator.divide(denominator).loe(maxRatio)); } return condition; } }

CourseQueryRepository 통계 쿼리

1. 강의 전체 통계

// 강의 전체 통계 public CourseStatistics getCourseStatistics() { return queryFactory .select(Projections.constructor(CourseStatistics.class, course.countDistinct(), course.currentStudents.sum(), course.maxStudents.sum(), course.currentStudents.avg(), course.maxStudents.avg() )) .from(course) .where(CommonQueryConditions.notDeleted(course.deleted)) .fetchOne(); }

2. 강의별 상세 통계

// 강의별 상세 통계 (수강률 포함) public List<CourseStatistics> getCourseDetailedStatistics() { return queryFactory .select(Projections.constructor(CourseStatistics.class, course.courseNo, course.name, course.instructor.fullName, course.maxStudents, course.currentStudents, Expressions.stringTemplate("CASE WHEN {0} > 0 THEN CAST({1} AS double) / {0} ELSE 0 END", course.maxStudents, course.currentStudents).as("enrollmentRatio") )) .from(course) .join(course.instructor, user) .where(CommonQueryConditions.notDeleted(course.deleted)) .orderBy(course.name.asc()) .fetch(); }

3. 강사별 강의 통계

// 강사별 강의 통계 public List<CourseStatistics> getInstructorStatistics() { return queryFactory .select(Projections.constructor(CourseStatistics.class, course.instructor.fullName, course.countDistinct(), course.currentStudents.sum(), course.maxStudents.sum() )) .from(course) .join(course.instructor, user) .where(CommonQueryConditions.notDeleted(course.deleted)) .groupBy(course.instructor.fullName) .orderBy(course.instructor.fullName.asc()) .fetch(); }

4. 월별 강의 개설 통계

// 월별 강의 개설 통계 public List<CourseStatistics> getMonthlyCourseStatistics() { return queryFactory .select(Projections.constructor(CourseStatistics.class, Expressions.stringTemplate("DATE_FORMAT({0}, '%Y-%m')", course.createdAt).as("month"), course.count() )) .from(course) .where(CommonQueryConditions.notDeleted(course.deleted)) .groupBy(Expressions.stringTemplate("DATE_FORMAT({0}, '%Y-%m')", course.createdAt)) .orderBy(Expressions.stringTemplate("DATE_FORMAT({0}, '%Y-%m')", course.createdAt).asc()) .fetch(); }

동적 검색 기능

1. 강의 텍스트 검색

// 강의 텍스트 검색 (강의명, 설명, 강사명) public Page<Course> searchCoursesByText(String searchText, Pageable pageable) { BooleanExpression predicate = CommonQueryConditions.notDeleted(course.deleted); if (searchText != null && !searchText.trim().isEmpty()) { String trimmedText = searchText.trim(); predicate = predicate.and( course.name.contains(trimmedText) .or(course.description.contains(trimmedText)) .or(course.instructor.fullName.contains(trimmedText)) ); } List<Course> content = queryFactory .selectFrom(course) .join(course.instructor, user).fetchJoin() .where(predicate) .offset(pageable.getOffset()) .limit(pageable.getPageSize()) .orderBy(createOrderSpecifier(pageable.getSort())) .fetch(); Long total = queryFactory .select(course.count()) .from(course) .where(predicate) .fetchOne(); return new PageImpl<>(content, pageable, total != null ? total : 0); }

2. 강의 고급 검색

// 강의 고급 검색 (CourseSearchCondition DTO 사용) public Page<Course> searchCourses(CourseSearchCondition condition, Pageable pageable) { List<Course> content = queryFactory .selectFrom(course) .join(course.instructor, user).fetchJoin() .where( CommonQueryConditions.notDeleted(course.deleted), CommonQueryConditions.stringContains(course.name, condition.getCourseName()), CommonQueryConditions.stringContains(course.instructor.fullName, condition.getInstructorName()), CommonQueryConditions.stringContains(course.description, condition.getDescription()), CommonQueryConditions.numberBetween(course.maxStudents, condition.getMinCapacity(), condition.getMaxCapacity()), CommonQueryConditions.numberBetween(course.currentStudents, condition.getMinCurrentStudents(), condition.getMaxCurrentStudents()), isAvailable(condition.getIsAvailable()), CommonQueryConditions.ratioBetween(course.currentStudents, course.maxStudents, condition.getMinEnrollmentRatio(), null), CommonQueryConditions.dateBetween(course.createdAt, condition.getStartDate(), condition.getEndDate()) ) .offset(pageable.getOffset()) .limit(pageable.getPageSize()) .orderBy(createOrderSpecifier(pageable.getSort())) .fetch(); Long total = queryFactory .select(course.count()) .from(course) .where( CommonQueryConditions.notDeleted(course.deleted), CommonQueryConditions.stringContains(course.name, condition.getCourseName()), CommonQueryConditions.stringContains(course.instructor.fullName, condition.getInstructorName()), CommonQueryConditions.stringContains(course.description, condition.getDescription()), CommonQueryConditions.numberBetween(course.maxStudents, condition.getMinCapacity(), condition.getMaxCapacity()), CommonQueryConditions.numberBetween(course.currentStudents, condition.getMinCurrentStudents(), condition.getMaxCurrentStudents()), isAvailable(condition.getIsAvailable()), CommonQueryConditions.ratioBetween(course.currentStudents, course.maxStudents, condition.getMinEnrollmentRatio(), null), CommonQueryConditions.dateBetween(course.createdAt, condition.getStartDate(), condition.getEndDate()) ) .fetchOne(); return new PageImpl<>(content, pageable, total != null ? total : 0); }
